1. Basic Retrieval
SELECT * FROM Equipment;

2. Filtering
SELECT * FROM Equipment WHERE IsAvailable = 1;

3. Join
SELECT
    s.Name AS Student,
    e.Name AS Equipment,
    b.DateBorrowed AS Borrowed,
    b.ExpectedReturnDate AS Due
FROM Borrowings b
INNER JOIN Students s ON b.StudentId = s.StudentId
INNER JOIN Equipment e ON b.EquipmentId = e.EquipmentId
WHERE b.Status = 0; -- 0 = Active

 4. Aggregate
SELECT COUNT(*) AS ActiveBorrowingCount
FROM Borrowings
WHERE Status = 0;

5. Update
UPDATE Equipment
SET IsAvailable = 1
WHERE EquipmentId = 2;
